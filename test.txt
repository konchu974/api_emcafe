# ========================================
# ÉTAPE 1 : Connexion
# ========================================
Write-Host "🔐 ÉTAPE 1 : Connexion à l'API EMCA" -ForegroundColor Yellow
Write-Host "─────────────────────────────────────────────────────" -ForegroundColor Gray

try {
    $loginBody = @{
        email = "Jean.Bonot@example.com"      # ⚠️ MODIFIER ICI
        password = "motdepasse123" # ⚠️ MODIFIER ICI
    } | ConvertTo-Json

    $loginResponse = Invoke-RestMethod `
        -Uri "https://api-emcafe-3.onrender.com/api/auth/login" `
        -Method Post `
        -ContentType "application/json; charset=utf-8" `
        -Body ([System.Text.Encoding]::UTF8.GetBytes($loginBody))

    $token = $loginResponse.token
    
    Write-Host "✅ Connexion réussie !" -ForegroundColor Green
    Write-Host "   Token: $($token.Substring(0,50))..." -ForegroundColor Gray
    Write-Host "   User : $($loginResponse.user.email)" -ForegroundColor White
    Write-Host ""

} catch {
    Write-Host "❌ ERREUR de connexion" -ForegroundColor Red
    Write-Host "Message: $($_.Exception.Message)" -ForegroundColor Red
    
    if ($_.Exception.Response) {
        $result = $_.Exception.Response.GetResponseStream()
        $reader = New-Object System.IO.StreamReader($result)
        $body = $reader.ReadToEnd()
        Write-Host "Détails: $body" -ForegroundColor Gray
    }
    exit
}

# ========================================
# ÉTAPE 2 : Récupérer une commande
# ========================================
Write-Host "📦 ÉTAPE 2 : Récupération d'une commande" -ForegroundColor Yellow
Write-Host "─────────────────────────────────────────────────────" -ForegroundColor Gray

try {
    $ordersResponse = Invoke-RestMethod `
        -Uri "https://api-emcafe-3.onrender.com/api/orders?limit=1" `
        -Method Get `
        -Headers @{ Authorization = "Bearer $token" }
    
    if ($ordersResponse.orders.Count -eq 0) {
        Write-Host "⚠️ Aucune commande trouvée - Création d'une commande test..." -ForegroundColor Yellow
        
        # Créer une commande test
        $newOrderBody = @{
            items = @(
                @{
                    product_id = 1
                    quantity = 2
                    price = 12.90
                }
            )
            shipping_address = @{
                name = "Test Client"
                address = "10 Rue de Test"
                city = "Paris"
                postal_code = "75001"
                country = "FR"
                phone = "+33612345678"
                email = "test@example.com"
            }
        } | ConvertTo-Json -Depth 5
        
        $createOrderResponse = Invoke-RestMethod `
            -Uri "https://api-emcafe-3.onrender.com/api/orders" `
            -Method Post `
            -Headers @{ 
                Authorization = "Bearer $token"
                "Content-Type" = "application/json; charset=utf-8"
            } `
            -Body ([System.Text.Encoding]::UTF8.GetBytes($newOrderBody))
        
        $orderId = $createOrderResponse.order.id_order
        Write-Host "✅ Commande créée : #$orderId" -ForegroundColor Green
    } else {
        $orderId = $ordersResponse.orders[0].id_order
        Write-Host "✅ Commande trouvée : #$orderId" -ForegroundColor Green
    }
    
    Write-Host ""

} catch {
    Write-Host "❌ ERREUR récupération commande" -ForegroundColor Red
    Write-Host "Message: $($_.Exception.Message)" -ForegroundColor Red
    exit
}

# ========================================
# ÉTAPE 3 : Créer le colis SendCloud
# ========================================
Write-Host "📦 ÉTAPE 3 : Création du colis SendCloud" -ForegroundColor Yellow
Write-Host "─────────────────────────────────────────────────────" -ForegroundColor Gray

try {
    $parcelBody = @{
        order_id = $orderId
        recipient = @{
            name = "Sophie Martin"
            email = "sophie.martin@example.com"
            phone = "+33612345678"
            address = "15 Boulevard Haussmann"
            city = "Paris"
            postal_code = "75009"
            country = "FR"
        }
        items = @(
            @{
                description = "Café Arabica Bio 250g"
                quantity = 2
                weight = 0.250
                value = 12.90
                hs_code = "09011100"
                origin_country = "FR"
            },
            @{
                description = "Café Robusta 500g"
                quantity = 1
                weight = 0.500
                value = 18.50
                hs_code = "09011200"
                origin_country = "FR"
            }
        )
        weight = 1.0
    } | ConvertTo-Json -Depth 10

    Write-Host "📄 Payload envoyé:" -ForegroundColor Gray
    Write-Host $parcelBody -ForegroundColor DarkGray
    Write-Host ""

    $parcelResponse = Invoke-RestMethod `
        -Uri "https://api-emcafe-3.onrender.com/api/sendcloud/parcels" `
        -Method Post `
        -Headers @{ 
            Authorization = "Bearer $token"
            "Content-Type" = "application/json; charset=utf-8"
        } `
        -Body ([System.Text.Encoding]::UTF8.GetBytes($parcelBody))

    Write-Host "✅ ✅ ✅ COLIS CRÉÉ AVEC SUCCÈS !" -ForegroundColor Green
    Write-Host "═══════════════════════════════════════════════════════" -ForegroundColor Green
    Write-Host "📦 ID SendCloud  : $($parcelResponse.data.parcel_id)" -ForegroundColor Cyan
    Write-Host "📋 N° Tracking   : $($parcelResponse.data.tracking_number)" -ForegroundColor Yellow
    Write-Host "🔗 URL Suivi     : $($parcelResponse.data.tracking_url)" -ForegroundColor Blue
    Write-Host "🏷️  URL Étiquette : $($parcelResponse.data.label_url)" -ForegroundColor Magenta
    Write-Host "📦 ID Commande   : $($parcelResponse.data.order_id)" -ForegroundColor White
    Write-Host "═══════════════════════════════════════════════════════" -ForegroundColor Green
    Write-Host ""

} catch {
    Write-Host "❌ ERREUR création colis" -ForegroundColor Red
    Write-Host "─────────────────────────────────────────────────────" -ForegroundColor Gray
    Write-Host "Message: $($_.Exception.Message)" -ForegroundColor Red
    
    if ($_.Exception.Response) {
        try {
            $result = $_.Exception.Response.GetResponseStream()
            $reader = New-Object System.IO.StreamReader($result)
            $body = $reader.ReadToEnd()
            $reader.Close()
            $result.Close()
            
            Write-Host "`n📋 Réponse du serveur:" -ForegroundColor Yellow
            try {
                $errorJson = $body | ConvertFrom-Json
                $errorJson | ConvertTo-Json -Depth 5 | Write-Host -ForegroundColor Red
            } catch {
                Write-Host $body -ForegroundColor Red
            }
        } catch {
            Write-Host "Impossible de lire la réponse" -ForegroundColor Gray
        }
    }
    Write-Host ""
}
