# ========================================
# 🧪 TEST COMPLET API SENDCLOUD - VERSION CORRIGÉE
# ========================================

$ErrorActionPreference = "Stop"
$baseUrl = "http://localhost:3000"

# Codes couleur
function Write-Success { Write-Host "✅ $args" -ForegroundColor Green }
function Write-Error { Write-Host "❌ $args" -ForegroundColor Red }
function Write-Info { Write-Host "ℹ️  $args" -ForegroundColor Cyan }
function Write-Warning { Write-Host "⚠️  $args" -ForegroundColor Yellow }
function Write-Step { 
    Write-Host "`n========================================" -ForegroundColor Magenta
    Write-Host "🔹 $args" -ForegroundColor Magenta
    Write-Host "========================================" -ForegroundColor Magenta 
}

# Variables globales
$script:authToken = $null
$script:parcelId = $null

# ========================================
# 🔐 CONFIGURATION UTILISATEUR
# ========================================
Write-Step "🔐 CONFIGURATION"

Write-Warning "Quel utilisateur voulez-vous utiliser ?"
Write-Host "1) Créer un nouvel utilisateur admin"
Write-Host "2) Utiliser l'utilisateur invité (invite@invite.com)"
Write-Host "3) Utiliser mes identifiants personnalisés"
$choice = Read-Host "Votre choix (1/2/3)"

switch ($choice) {
    "1" {
        # Créer un admin
        Write-Info "Création d'un utilisateur admin..."
        $adminPayload = @{
            email = "admin@emcafe.fr"
            password = "Admin123!"
            firstName = "Admin"
            lastName = "System"
            role = "ADMIN"
        } | ConvertTo-Json
        
        try {
            $registerResponse = Invoke-RestMethod -Uri "$baseUrl/api/auth/register" `
                -Method Post `
                -ContentType "application/json" `
                -Body $adminPayload
            
            Write-Success "Admin créé: admin@emcafe.fr"
            $script:userEmail = "admin@emcafe.fr"
            $script:userPassword = "Admin123!"
        } catch {
            Write-Warning "Admin existe déjà, connexion..."
            $script:userEmail = "admin@emcafe.fr"
            $script:userPassword = "Admin123!"
        }
    }
    "2" {
        # Utiliser l'invité
        Write-Info "Utilisation du compte invité..."
        $script:userEmail = "invite@invite.com"
        $script:userPassword = "Passing3-Scuba4-Durably6-Judge6"
    }
    "3" {
        # Identifiants personnalisés
        $script:userEmail = Read-Host "Email"
        $script:userPassword = Read-Host "Mot de passe" -AsSecureString
        $script:userPassword = [Runtime.InteropServices.Marshal]::PtrToStringAuto(
            [Runtime.InteropServices.Marshal]::SecureStringToBSTR($script:userPassword)
        )
    }
    default {
        Write-Error "Choix invalide"
        exit 1
    }
}

# ========================================
# 1️⃣ HEALTH CHECK
# ========================================
Write-Step "1️⃣ HEALTH CHECK"

try {
    $health = Invoke-RestMethod -Uri "$baseUrl/health" -Method Get
    Write-Success "Server is running"
    Write-Info "Timestamp: $($health.timestamp)"
} catch {
    Write-Error "Server is not running! Start it with: npm run dev"
    exit 1
}

# ========================================
# 2️⃣ AUTHENTIFICATION
# ========================================
Write-Step "2️⃣ AUTHENTIFICATION"

$loginPayload = @{
    email = $script:userEmail
    password = $script:userPassword
} | ConvertTo-Json

try {
    Write-Info "Connexion avec: $script:userEmail"
    
    $authResponse = Invoke-RestMethod -Uri "$baseUrl/api/auth/login" `
        -Method Post `
        -ContentType "application/json" `
        -Body $loginPayload
    
    $script:authToken = $authResponse.token
    Write-Success "Authentification réussie"
    Write-Info "Token: $($script:authToken.Substring(0, 20))..."
    Write-Info "User ID: $($authResponse.user.id_user_account)"
} catch {
    Write-Error "Authentification échouée"
    Write-Error "Détails: $($_.Exception.Message)"
    Write-Warning "Vérifiez que l'utilisateur existe dans la base de données"
    exit 1
}

# Headers avec authentification
$headers = @{
    "Authorization" = "Bearer $script:authToken"
    "Content-Type" = "application/json"
}

# ========================================
# 3️⃣ RÉCUPÉRATION DES MÉTHODES D'EXPÉDITION
# ========================================
Write-Step "3️⃣ RÉCUPÉRATION DES MÉTHODES D'EXPÉDITION"

try {
    $shippingMethods = Invoke-RestMethod -Uri "$baseUrl/api/sendcloud/shipping-methods" `
        -Method Get `
        -Headers $headers
    
    if ($shippingMethods.success) {
        Write-Success "$($shippingMethods.count) méthode(s) d'expédition disponible(s)"
        
        # Afficher les 5 premières
        $shippingMethods.data | Select-Object -First 5 | ForEach-Object {
            Write-Info "  - [$($_.id)] $($_.name) ($($_.carrier))"
        }
        
        # Sauvegarder une méthode pour la suite
        if ($shippingMethods.data.Count -gt 0) {
            $script:defaultShippingMethod = $shippingMethods.data[0].id
            Write-Info "Méthode par défaut: $script:defaultShippingMethod"
        }
    } else {
        Write-Warning "Aucune méthode d'expédition configurée dans SendCloud"
    }
} catch {
    Write-Error "Erreur récupération méthodes: $($_.Exception.Message)"
    Write-Warning "Continuez quand même..."
}

# ========================================
# 4️⃣ RECHERCHE DE POINTS RELAIS
# ========================================
Write-Step "4️⃣ RECHERCHE DE POINTS RELAIS"

try {
    $servicePoints = Invoke-RestMethod -Uri "$baseUrl/api/sendcloud/service-points?postal_code=75009&country=FR" `
        -Method Get
    
    if ($servicePoints.success) {
        Write-Success "$($servicePoints.count) point(s) relais trouvé(s)"
        
        # Afficher les 3 premiers
        $servicePoints.data | Select-Object -First 3 | ForEach-Object {
            Write-Info "  - [$($_.id)] $($_.name)"
            Write-Info "    📍 $($_.address), $($_.postal_code) $($_.city)"
            if ($_.distance) {
                Write-Info "    📏 Distance: $($_.distance) m"
            }
        }
        
        # Sauvegarder un point relais pour la suite
        if ($servicePoints.data.Count -gt 0) {
            $script:relayPointId = $servicePoints.data[0].id
            Write-Info "Point relais sélectionné: $script:relayPointId"
        }
    } else {
        Write-Warning "Aucun point relais trouvé"
    }
} catch {
    Write-Error "Erreur recherche points relais: $($_.Exception.Message)"
}

# ========================================
# 5️⃣ CRÉATION D'UN COLIS (LIVRAISON À DOMICILE)
# ========================================
Write-Step "5️⃣ CRÉATION D'UN COLIS (LIVRAISON À DOMICILE)"

$parcelPayload = @{
    name = "Jean Dupont"
    company_name = "Entreprise Test"
    address = "123 Rue de la Paix"
    address_2 = "Appartement 4B"
    city = "Paris"
    postal_code = "75009"
    country = "FR"
    email = "jean.dupont@example.com"
    telephone = "+33612345678"
    weight = "1000"
    order_number = "TEST-$(Get-Date -Format 'yyyyMMddHHmmss')"
    request_label = $true
    apply_shipping_rules = $true
    items = @(
        @{
            description = "Café Arabica Premium"
            quantity = 2
            weight = "500"
            value = "29.90"
            hs_code = "0901.21"
            origin_country = "FR"
            sku = "CAFE-ARABICA-001"
        }
    )
} | ConvertTo-Json -Depth 10

try {
    Write-Info "Envoi de la requête de création..."
    
    $parcelResponse = Invoke-RestMethod -Uri "$baseUrl/api/sendcloud/parcels" `
        -Method Post `
        -Headers $headers `
        -Body $parcelPayload
    
    if ($parcelResponse.success) {
        $script:parcelId = $parcelResponse.data.id
        Write-Success "Colis créé avec succès"
        Write-Info "  🆔 ID: $($parcelResponse.data.id)"
        Write-Info "  📦 Numéro de suivi: $($parcelResponse.data.tracking_number)"
        Write-Info "  🔗 URL de suivi: $($parcelResponse.data.tracking_url)"
        Write-Info "  📋 Statut: $($parcelResponse.data.status.message)"
        Write-Info "  🚚 Transporteur: $($parcelResponse.data.carrier.name)"
    } else {
        Write-Error "Échec de la création: $($parcelResponse.error)"
    }
} catch {
    Write-Error "Erreur création colis: $($_.Exception.Message)"
    Write-Info "Réponse brute: $($_.ErrorDetails.Message)"
}

# ========================================
# 📊 RÉSUMÉ FINAL
# ========================================
Write-Step "📊 RÉSUMÉ FINAL"

Write-Success "Tests terminés !"
Write-Info ""
Write-Info "👤 Utilisateur: $script:userEmail"
if ($script:parcelId) {
    Write-Info "📦 Colis créé: ID $script:parcelId"
}
Write-Info ""
Write-Info "🔗 Dashboard SendCloud:"
Write-Info "   https://panel.sendcloud.sc/parcels"
Write-Info ""
Write-Success "✨ Backend SendCloud opérationnel !"
