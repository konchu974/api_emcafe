# ========================================
# 🧪 VERSION CORRIGÉE - SANS ConvertTo-Utf8Json
# ========================================

$ErrorActionPreference = "Stop"

# ⚡ FORCER UTF-8 GLOBALEMENT
[Console]::OutputEncoding = [System.Text.Encoding]::UTF8
$PSDefaultParameterValues['*:Encoding'] = 'utf8'

$baseUrl = "https://api-emcafe-3.onrender.com"

# Codes couleur
function Write-Success { Write-Host "✅ $args" -ForegroundColor Green }
function Write-Error { Write-Host "❌ $args" -ForegroundColor Red }
function Write-Info { Write-Host "ℹ️  $args" -ForegroundColor Cyan }
function Write-Warning { Write-Host "⚠️  $args" -ForegroundColor Yellow }
function Write-Step { 
    Write-Host "`n========================================" -ForegroundColor Magenta
    Write-Host "🔹 $args" -ForegroundColor Magenta
    Write-Host "========================================" -ForegroundColor Magenta 
}

# Variables globales
$script:authToken = $null
$script:parcelId = $null
$script:userEmail = $null
$script:userPassword = $null
$script:defaultShippingMethod = $null
$script:relayPointId = $null

# ========================================
# 🔐 CONFIGURATION UTILISATEUR
# ========================================
Write-Step "🔐 CONFIGURATION"

Write-Warning "Quel utilisateur voulez-vous utiliser ?"
Write-Host "1) Créer un nouvel utilisateur admin"
Write-Host "2) Utiliser l'utilisateur invité (invite@invite.com)"
Write-Host "3) Utiliser mes identifiants personnalisés"
$choice = Read-Host "Votre choix (1/2/3)"

switch ($choice) {
    "1" {
        # Créer un admin
        Write-Info "Création d'un utilisateur admin..."
        $adminPayload = @{
            email = "admin@emcafe.fr"
            password = "Admin123!"
            first_name = "Admin"
            last_name = "System"
            phone = "+33612345678"
            role = "ADMIN"
        } | ConvertTo-Json -Depth 10
        
        # ⚠️ SANS -Compress pour debug
        Write-Info "Payload envoyé:"
        Write-Host $adminPayload -ForegroundColor Gray
        
        try {
            $registerResponse = Invoke-RestMethod -Uri "$baseUrl/api/auth/register" `
                -Method Post `
                -ContentType "application/json; charset=utf-8" `
                -Body $adminPayload
            
            Write-Success "Admin créé: admin@emcafe.fr"
            $script:userEmail = "admin@emcafe.fr"
            $script:userPassword = "Admin123!"
        } catch {
            $statusCode = $_.Exception.Response.StatusCode.value__
            if ($statusCode -eq 409) {
                Write-Warning "Admin existe déjà, connexion..."
                $script:userEmail = "admin@emcafe.fr"
                $script:userPassword = "Admin123!"
            } else {
                Write-Error "Erreur $statusCode : $($_.Exception.Message)"
                if ($_.ErrorDetails.Message) {
                    $errorDetails = $_.ErrorDetails.Message | ConvertFrom-Json
                    Write-Info "Détails:"
                    $errorDetails | ConvertTo-Json -Depth 5 | Write-Host -ForegroundColor Yellow
                }
                exit 1
            }
        }
    }
    "2" {
        Write-Info "Utilisation du compte invité..."
        $script:userEmail = "invite@invite.com"
        $script:userPassword = "Passing3-Scuba4-Durably6-Judge6"
    }
    "3" {
        $script:userEmail = Read-Host "Email"
        $securePassword = Read-Host "Mot de passe" -AsSecureString
        $script:userPassword = [Runtime.InteropServices.Marshal]::PtrToStringAuto(
            [Runtime.InteropServices.Marshal]::SecureStringToBSTR($securePassword)
        )
    }
    default {
        Write-Error "Choix invalide"
        exit 1
    }
}

# ========================================
# 1️⃣ HEALTH CHECK
# ========================================
Write-Step "1️⃣ HEALTH CHECK"

try {
    $health = Invoke-RestMethod -Uri "$baseUrl/health" -Method Get -TimeoutSec 10
    Write-Success "Server is running"
    Write-Info "Timestamp: $($health.timestamp)"
} catch {
    Write-Error "Server is not running!"
    Write-Info "URL testée: $baseUrl/health"
    Write-Info "Erreur: $($_.Exception.Message)"
    exit 1
}

# ========================================
# 2️⃣ AUTHENTIFICATION
# ========================================
Write-Step "2️⃣ AUTHENTIFICATION"

$loginPayload = @{
    email = $script:userEmail
    password = $script:userPassword
} | ConvertTo-Json -Depth 10

Write-Info "Payload login:"
Write-Host $loginPayload -ForegroundColor Gray

try {
    Write-Info "Connexion avec: $script:userEmail"
    
    $authResponse = Invoke-RestMethod -Uri "$baseUrl/api/auth/login" `
        -Method Post `
        -ContentType "application/json; charset=utf-8" `
        -Body $loginPayload `
        -TimeoutSec 15
    
    $script:authToken = $authResponse.token
    Write-Success "Authentification réussie"
    Write-Info "Token: $($script:authToken.Substring(0, 20))..."
    Write-Info "User ID: $($authResponse.user.id_user_account)"
    Write-Info "Role: $($authResponse.user.role)"
} catch {
    Write-Error "Authentification échouée"
    
    if ($_.ErrorDetails.Message) {
        try {
            $errorDetails = $_.ErrorDetails.Message | ConvertFrom-Json
            Write-Info "Message: $($errorDetails.message)"
            if ($errorDetails.errors) {
                Write-Info "Erreurs:"
                $errorDetails.errors | ForEach-Object { Write-Info "  - $_" }
            }
        } catch {
            Write-Info "Réponse: $($_.ErrorDetails.Message)"
        }
    } else {
        Write-Info "Erreur: $($_.Exception.Message)"
    }
    
    exit 1
}

# Headers avec authentification
$headers = @{
    "Authorization" = "Bearer $script:authToken"
    "Content-Type" = "application/json; charset=utf-8"
    "Accept" = "application/json"
}

# ========================================
# 3️⃣ RÉCUPÉRATION DES MÉTHODES D'EXPÉDITION
# ========================================
Write-Step "3️⃣ RÉCUPÉRATION DES MÉTHODES D'EXPÉDITION"

try {
    $shippingMethods = Invoke-RestMethod -Uri "$baseUrl/api/sendcloud/shipping-methods" `
        -Method Get `
        -Headers $headers `
        -TimeoutSec 15
    
    if ($shippingMethods.success) {
        Write-Success "$($shippingMethods.count) méthode(s) d'expédition disponible(s)"
        
        $shippingMethods.data | Select-Object -First 5 | ForEach-Object {
            Write-Info "  - [$($_.id)] $($_.name) ($($_.carrier))"
            if ($_.price) {
                Write-Info "    💰 Prix: $($_.price.currency) $($_.price.value)"
            }
        }
        
        if ($shippingMethods.data.Count -gt 0) {
            $script:defaultShippingMethod = $shippingMethods.data[0].id
            Write-Info "Méthode par défaut: $script:defaultShippingMethod"
        }
    } else {
        Write-Warning "Aucune méthode d'expédition configurée"
    }
} catch {
    Write-Error "Erreur récupération méthodes: $($_.Exception.Message)"
    if ($_.ErrorDetails.Message) {
        Write-Info "Détails: $($_.ErrorDetails.Message)"
    }
}

# ========================================
# 4️⃣ RECHERCHE DE POINTS RELAIS
# ========================================
Write-Step "4️⃣ RECHERCHE DE POINTS RELAIS"

try {
    $servicePointsUrl = "$baseUrl/api/sendcloud/service-points?postal_code=75009&country=FR"
    
    $servicePoints = Invoke-RestMethod -Uri $servicePointsUrl `
        -Method Get `
        -Headers @{ 
            "Authorization" = "Bearer $script:authToken"
            "Accept" = "application/json"
        } `
        -TimeoutSec 15
    
    if ($servicePoints.success) {
        Write-Success "$($servicePoints.count) point(s) relais trouvé(s)"
        
        $servicePoints.data | Select-Object -First 3 | ForEach-Object {
            Write-Info "  - [$($_.id)] $($_.name)"
            Write-Info "    📍 $($_.address), $($_.postal_code) $($_.city)"
            if ($_.distance) {
                Write-Info "    📏 Distance: $([math]::Round($_.distance, 0)) m"
            }
        }
        
        if ($servicePoints.data.Count -gt 0) {
            $script:relayPointId = $servicePoints.data[0].id
            Write-Info "Point relais sélectionné: $script:relayPointId"
        }
    } else {
        Write-Warning "Aucun point relais trouvé"
    }
} catch {
    Write-Error "Erreur recherche points relais: $($_.Exception.Message)"
    if ($_.ErrorDetails.Message) {
        Write-Info "Détails: $($_.ErrorDetails.Message)"
    }
}

# ========================================
# 5️⃣ CRÉATION D'UN COLIS
# ========================================
Write-Step "5️⃣ CRÉATION D'UN COLIS"

$orderNumber = "TEST-$(Get-Date -Format 'yyyyMMddHHmmss')"
$parcelPayload = @{
    name = "Jean Dupont"
    company_name = "Entreprise Test"
    address = "123 Rue de la Paix"
    address_2 = "Appartement 4B"
    city = "Paris"
    postal_code = "75009"
    country = "FR"
    email = "jean.dupont@example.com"
    telephone = "+33612345678"
    weight = "1000"
    order_number = $orderNumber
    request_label = $true
    apply_shipping_rules = $true
    items = @(
        @{
            description = "Café Arabica Premium"
            quantity = 2
            weight = "500"
            value = "29.90"
            hs_code = "0901.21"
            origin_country = "FR"
            sku = "CAFE-ARABICA-001"
        }
    )
} | ConvertTo-Json -Depth 10

Write-Info "Payload parcel:"
Write-Host $parcelPayload -ForegroundColor Gray

try {
    Write-Info "Création du colis..."
    
    $parcelResponse = Invoke-RestMethod -Uri "$baseUrl/api/sendcloud/parcels" `
        -Method Post `
        -Headers $headers `
        -Body $parcelPayload `
        -TimeoutSec 30
    
    if ($parcelResponse.success) {
        $script:parcelId = $parcelResponse.data.id
        Write-Success "Colis créé!"
        Write-Info "  🆔 ID: $($parcelResponse.data.id)"
        Write-Info "  📦 Tracking: $($parcelResponse.data.tracking_number)"
        Write-Info "  🔗 URL: $($parcelResponse.data.tracking_url)"
        Write-Info "  📋 Statut: $($parcelResponse.data.status.message)"
        Write-Info "  🚚 Carrier: $($parcelResponse.data.carrier.name)"
        
        if ($parcelResponse.data.label) {
            Write-Info "  🏷️  Label: $($parcelResponse.data.label.label_printer)"
        }
    } else {
        Write-Error "Échec création colis"
        Write-Info "Message: $($parcelResponse.error)"
    }
} catch {
    Write-Error "Erreur création colis: $($_.Exception.Message)"
    
    if ($_.ErrorDetails.Message) {
        try {
            $errorDetails = $_.ErrorDetails.Message | ConvertFrom-Json
            Write-Info "Réponse:"
            $errorDetails | ConvertTo-Json -Depth 5 | Write-Host -ForegroundColor Yellow
        } catch {
            Write-Info "Réponse brute: $($_.ErrorDetails.Message)"
        }
    }
}

# ========================================
# 📊 RÉSUMÉ FINAL
# ========================================
Write-Step "📊 RÉSUMÉ FINAL"

Write-Host ""
Write-Success "═══════════════════════════════════════════════"
Write-Success "  ✨ Tests terminés!"
Write-Success "═══════════════════════════════════════════════"
Write-Host ""

Write-Info "👤 Compte: $script:userEmail"
if ($script:authToken) {
    Write-Host "   Token: $($script:authToken.Substring(0, 30))..." -ForegroundColor Gray
}

if ($script:parcelId) {
    Write-Info "📦 Colis: $script:parcelId"
    Write-Host "   Commande: $orderNumber" -ForegroundColor White
}

if ($script:defaultShippingMethod) {
    Write-Info "🚚 Méthode: $script:defaultShippingMethod"
}

if ($script:relayPointId) {
    Write-Info "📍 Point relais: $script:relayPointId"
}

Write-Host ""
Write-Info "🔗 Liens:"
Write-Host "   Dashboard: https://panel.sendcloud.sc/parcels" -ForegroundColor Cyan
Write-Host "   API Docs: $baseUrl/api-docs" -ForegroundColor Cyan
Write-Host ""

Write-Success "✅ Backend opérationnel!"
Write-Host ""
